  loadData(): void {
    this.loading = true;
      let model ={
        id: this.productId
    }
    this.httpService
      .customerPortalPosts('admin/customer/portal/view-customer-request', model)
      .subscribe((res: any) => {
        if (res.status === 200) {
          this.requestDetails = res.data
          this.loading = false;
          console.log(res.data);
        } else {
          this.loading = false;
        }
      });
    this.loading = false;
  }

  raiseAppeal(){
    this.modalRef = this.modalService.open(ConfirmDialogComponent, {centered: true});
    this.modalRef.componentInstance.title = 'RAISE AN APPEAL';
    this.modalRef.componentInstance.body= "Do you want to APPEAL for this Request?";
    this.modalRef.result.then((result) => {
      if (result === 'success') {
        this.httpService.customerPortalPost(`api/v1/portal/appeal/${this.productId}`, {}).subscribe(
          (result: any) => {
            if (result.status === "00") {
              Swal.fire('Appealed',  'Request Appealed Raised Successfully.',  'success')
              this.loadData()
            } else {

            }
          }
        );
        Swal.fire('Appealed',  'Request Appealed Raised Successfully.',  'success')
      } else {
        console.log("Error occurred")
      }
    });
  }
